# rule1

1. 不要有太多无用注释
2. 方法要分类, 不同类别之间要用public/protected/private分隔, 类别名称要放到分割符上面用/* xxx */来写

## example 1

### bad

```cpp
// Vulkan timeline semaphore backing for FgfxFence. Each instance owns a single
// VK_KHR_timeline_semaphore whose monotonic 64-bit value orders submissions and
// cross-queue ownership transfers. See knowledge-note/wgpu-multi-queue.md for
// the synchronization model: FgfxFence maps to timeline semaphore on Vulkan.
class VkFence : public VkResource, public Fence {
  FLUXEL_DOWNCAST_BASE_OBJECT(VkResource, Fence);

 public:
  // Construction
  explicit VkFence(intrusive_ptr<VkDevice> device);

  ~VkFence() override { vk_device_ref().handle().destroySemaphore(m_semaphore); }

 public:
  // Timeline operations

  // Returns the last signaled value the device has observed.
  uint64_t query_completed_value() const { return vk_device_ref().handle().getSemaphoreCounterValue(m_semaphore); }

  // Block the host until the counter reaches or exceeds `value`.
  // `timeout_ns` == UINT64_MAX means wait indefinitely. Returns false on timeout.
  bool wait(uint64_t value, uint64_t timeout_ns) const;

  // Advance the counter to `value` from the host. Used internally by submit.
  void signal(uint64_t value);

 public:
  [[nodiscard]] vk::Semaphore vk_semaphore() const noexcept { return m_semaphore; }

 protected:
  vk::Semaphore m_semaphore;
};
```

### good

```cpp
class VkFence : public VkResource, public Fence {
  FLUXEL_DOWNCAST_BASE_OBJECT(VkResource, Fence);

 public:
  explicit VkFence(intrusive_ptr<VkDevice> device);

  ~VkFence() override { vk_device_ref().handle().destroySemaphore(m_semaphore); }

 /* Timeline operatoion */
 public:
  uint64_t query_completed_value() const { return vk_device_ref().handle().getSemaphoreCounterValue(m_semaphore); }

  bool wait(uint64_t value, uint64_t timeout_ns) const;

  void signal(uint64_t value);

 public:
  vk::Semaphore vk_semaphore() const noexcept { return m_semaphore; }

 protected:
  vk::Semaphore m_semaphore;
};
```

## example 2

### bad

```cpp
// Backend-neutral base for a finished command buffer. A concrete backend owns
// the native command buffer (and the pool it was allocated from) extracted from
// the encoder on finish(); it is freed once the GPU has finished executing it.
class CommandBuffer : public Resource {
  FLUXEL_DOWNCAST_BASE_OBJECT(Resource);

 public:
  // Construction
  explicit CommandBuffer(intrusive_ptr<Device> device);

  ~CommandBuffer() override = default;

 public:
  // Accessors
  intrusive_ptr<Device> device() { return m_device; }

  intrusive_ptr<const Device> device() const { return m_device; }

  Device& device_ref() { return *m_device; }

  const Device& device_ref() const { return *m_device; }

 protected:
  intrusive_ptr<Device> m_device;
};
```

### good

```cpp
class CommandBuffer : public Resource {
  FLUXEL_DOWNCAST_BASE_OBJECT(Resource);

 public:
  explicit CommandBuffer(intrusive_ptr<Device> device);

  ~CommandBuffer() override = default;

 public:

  intrusive_ptr<Device> m_device;

```

## example 3

### bad

```cpp
std::vector<BufferBarrier> DeviceBufferTracker::set_from_tracker_and_drain(const BufferTracker& cb) {
  const auto& start = cb.start_state();
  const auto& end = cb.end_state();

  for (auto* buf : cb.tracked_buffers()) {
    if (!m_tracked.contains(buf)) {
      // Newly seen on the device side: adopt the command buffer's start state
      // as the device current state. No barrier is needed (nothing prior to
      // transition from).
      m_current.copy_buffer_state(buf, start);
      m_tracked.insert(buf);
      continue;
    }
    // Transition device current into the command buffer's start state, then
    // advance device current to the command buffer's end state.
    m_pending.append_range(start.query_barrier(buf, m_current, m_ordered_uses_mask));
    m_current.copy_buffer_state(buf, end);
  }

  return std::move(m_pending);
}
```

### good

```cpp
std::vector<BufferBarrier> DeviceBufferTracker::set_from_tracker_and_drain(const BufferTracker& cb) {
  const auto& start = cb.start_state();
  const auto& end = cb.end_state();

  for (auto* buf : cb.tracked_buffers()) {
    if (!m_tracked.contains(buf)) {
      m_current.copy_buffer_state(buf, start);
      m_tracked.insert(buf);
      continue;
    }
    m_pending.append_range(start.query_barrier(buf, m_current, m_ordered_uses_mask));
    m_current.copy_buffer_state(buf, end);
  }

  return std::move(m_pending);
}
```

# rule2

方法之间要用空行隔开

## bad

```cpp
class VkFence : public VkResource, public Fence {
 public:
  explicit VkFence(intrusive_ptr<VkDevice> device);
  ~VkFence();
}
```

## good

```cpp
class VkFence : public VkResource, public Fence {
 public:
  explicit VkFence(intrusive_ptr<VkDevice> device);

  ~VkFence();
}
```
